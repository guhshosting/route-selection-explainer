cmake_minimum_required(VERSION 3.24)
project(route-selection-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC (or clang-cl) are required")
endif()
option(RSE_ANALYZE "Enable MSVC static analysis" OFF)
option(RSE_ASAN "Enable AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rse_core src/address.cpp src/analysis.cpp src/windows_query.cpp src/format.cpp src/cli.cpp)
target_include_directories(rse_core PUBLIC include)
target_compile_definitions(rse_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(rse_core PUBLIC /W4 /permissive- /utf-8 /EHsc /external:anglebrackets /external:W0)
target_link_libraries(rse_core PUBLIC iphlpapi ws2_32)
if(RSE_ANALYZE)
  target_compile_options(rse_core PUBLIC /analyze /analyze:external-)
endif()
if(RSE_ASAN)
  target_compile_options(rse_core PUBLIC /fsanitize=address /Zi)
  target_link_options(rse_core PUBLIC /INCREMENTAL:NO)
endif()
add_executable(route-selection-explainer src/main.cpp)
target_link_libraries(route-selection-explainer PRIVATE rse_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(rse_tests tests/tests.cpp)
  target_link_libraries(rse_tests PRIVATE rse_core)
  add_test(NAME synthetic COMMAND rse_tests)
  add_test(NAME integration COMMAND rse_tests --integration)
  add_test(NAME version COMMAND route-selection-explainer --version)
  add_test(NAME help COMMAND route-selection-explainer --help)
  find_program(RSE_NODE node)
  if(RSE_NODE)
    add_test(NAME json_cli COMMAND "${RSE_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/json_cli.js" "$<TARGET_FILE:route-selection-explainer>" "$<TARGET_FILE:rse_tests>" "${CMAKE_CURRENT_SOURCE_DIR}/docs/json-schema.json")
  endif()
endif()
install(TARGETS route-selection-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE VALIDATION.md DESTINATION share/route-selection-explainer)
install(FILES docs/json-schema.json DESTINATION share/route-selection-explainer/docs)
